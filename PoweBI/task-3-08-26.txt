Suggested DAX Measures
Total Employees =
COUNT(Employee[EmpID])
Average Salary =
AVERAGE(Employee[Salary])
Attrition Count =
CALCULATE(
    COUNT(Employee[EmpID]),
    Employee[Attrition] = "Yes"
)
Attrition % =
DIVIDE(
    [Attrition Count],
    [Total Employees]
)
Average Experience =
AVERAGE(Employee[Experience])
Average Training Hours =
AVERAGE(Employee[TrainingHours])
